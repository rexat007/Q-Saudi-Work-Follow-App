import fs from 'node:fs';
const path='../closure-next-run-diagnostics.json';
export default class ClosureDiagnostics {
 data={};
 save(){fs.writeFileSync(path,JSON.stringify(this.data,null,2));}
 onInit(){process.once('exit',code=>{this.data.observedProcessExitCode=code;this.save();});}
 onTestRunEnd(_modules,errors,reason){this.data.reason=reason;this.data.exitCodeAtRunEnd=process.exitCode??0;this.data.unhandledErrors=errors.map(e=>Object.fromEntries(['name','message','stack','type','VITEST_TEST_NAME','VITEST_TEST_PATH'].filter(k=>e[k]!==undefined).map(k=>[k,e[k]])));this.save();}
 onProcessTimeout(){this.data.processTimeout=true;this.save();}
}
